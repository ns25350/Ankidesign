<script>
(function () {
  // FrontSideのスクリプトが実行されない場合にも文字サイズを整えます。
  var frontText = document.getElementById("front-text");
  if (frontText) {
    var frontLength = (frontText.textContent || "").trim().length;
    frontText.classList.toggle("text-small", frontLength > 20);
    frontText.classList.toggle("text-mid", frontLength > 10 && frontLength <= 20);
  }

  var slot = document.getElementById("answer-slot");
  var source = document.getElementById("back-content-source");
  if (slot && source) {
    while (slot.firstChild) slot.removeChild(slot.firstChild);
    var answer = source.querySelector(".answer-text");
    if (answer) {
      var length = (answer.textContent || "").replace(/\u200b/g, "").trim().length;
      var media = answer.querySelector("img, video, audio, iframe, svg, canvas, hr, table");
      if (length || media) {
        while (source.firstChild) slot.appendChild(source.firstChild);
      }
    }
    source.parentNode.removeChild(source);
  }
})();
</script>
