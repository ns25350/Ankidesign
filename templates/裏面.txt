<script>
(function () {
  // AnkiではFrontSide内のスクリプトが裏面で実行されない場合があるため、
  // 色の切替と文字サイズの設定を裏面でも初期化します。
  var themes = ["theme-orange", "theme-green", "theme-blue", "theme-pink"];

  window.changeAnkiTheme = function (theme) {
    if (themes.indexOf(theme) === -1) theme = "theme-orange";
    var card = document.querySelector(".card") || document.body;
    themes.forEach(function (name) { card.classList.remove(name); });
    card.classList.add(theme);
    document.querySelectorAll(".tab-btn").forEach(function (tab) {
      var active = tab.getAttribute("data-theme") === theme;
      tab.classList.toggle("active", active);
      tab.setAttribute("aria-pressed", active ? "true" : "false");
    });
    window.ankiCustomTheme = theme;
    try { localStorage.setItem("anki-custom-theme", theme); } catch (error) {}
  };

  var savedTheme = window.ankiCustomTheme || "theme-orange";
  try { savedTheme = localStorage.getItem("anki-custom-theme") || savedTheme; } catch (error) {}
  window.changeAnkiTheme(savedTheme);

  var frontText = document.getElementById("front-text");
  if (frontText) {
    var frontLength = (frontText.textContent || "").trim().length;
    frontText.classList.toggle("text-small", frontLength > 20);
    frontText.classList.toggle("text-mid", frontLength > 10 && frontLength <= 20);
  }

  var slot = document.getElementById("answer-slot");
  var content = document.getElementById("back-content-source");
  if (slot && content) {
    while (slot.firstChild) slot.removeChild(slot.firstChild);
    while (content.firstChild) slot.appendChild(content.firstChild);
    content.parentNode.removeChild(content);
    var answerLength = (slot.textContent || "").replace(/\u200b/g, "").trim().length;
    var media = slot.querySelector("img, video, audio, iframe, svg, canvas, hr, table");
    if (!answerLength && !media) slot.innerHTML = "";
    var structuredAnswer = slot.querySelector("ul, ol, table, pre, blockquote");
    slot.classList.toggle("text-long", answerLength > 40 || !!structuredAnswer);
  }
})();
</script>
